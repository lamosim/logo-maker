# Logo Maker Part 2: Sheets, Package and Naming Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add everything the owner sees and receives: the territory board, the concept sheet with in-context mockups, the refine sheet, the complete verified export package with print files and documentation, and the naming checks.

**Architecture:** Builds on Part 1's engine (`logomaker.kit`, `render.Renderer`, `audit`, `color`, `registry`). Sheets are Jinja templates with inline SVG, screenshotted by Chrome and opened with `open`. The package is built by a `PackageBuilder` whose groups each write files and register manifest entries; `verify` re-reads the manifest and checks every file. Print output is written by reportlab directly from kit paths. Naming checks use RDAP, `whois` and the GitHub API.

**Tech Stack:** Part 1's stack plus reportlab (PDF/EPS), pypdf (verification), httpx (RDAP, GitHub), Pillow (compositing, ICO, JPEG, WebP).

**Spec:** `docs/superpowers/specs/2026-10-05-logo-maker-design.md` (sections 4.2, 4.4, 4.5 step 5, 4.6 step 8, 4.7, 7.3, 9, 10, 13, 14). Platform numbers come from `docs/research/03-deliverable-spec.md`.

**Prerequisite:** Part 1 (`docs/superpowers/plans/2026-10-05-logo-maker-1-engine.md`) is implemented and its suites pass.

## Global Constraints

- Everything in Part 1's Global Constraints still applies (JSON stdout, exit codes 0/2/3/4, Chrome via Playwright channel chrome, paths-only SVG, determinism, `.env` denied, commit per task).
- File naming in the package: `{brand}-{variant}-{colour}[-{size}].{ext}`, lowercase, hyphenated. Variants: `primary`, `horizontal`, `stacked`, `wordmark`, `mark`, `small`, `favicon`. Colours: `color`, `reverse`, `black`, `white`, `mono`.
- `package build` may take up to 300 seconds (the skill sets the Bash timeout to 300000); every other command finishes within 60 seconds.
- Mockup neighbours (other app icons, other tabs) are generated by the kit; never draw or embed a third-party logo.
- Sheets open in the browser with `open <file>` unless `--no-open` is passed or the environment variable `LOGOMAKER_NO_OPEN` is set (tests set it).
- Maskable icons keep artwork inside the central circle of radius 40% of the canvas; avatars keep artwork inside an 80% circle; the YouTube banner keeps the logo inside the central 1235 by 338 px.
- Print PDFs are vector; CMYK values are a naive conversion and the README must say to confirm with the printer.

## Review Focus

1. A brand whose `final/` lacks `tokens.json` or `rationale.md`: `package build` must exit 2 naming the missing file, never produce a half package. Pinned in Task 15 (the builder checks before launching Chrome).
2. A wordmark-only brand (no symbol): sheets and exports that need `mark`, `small` or `favicon` must fall back to the wordmark and say so in warnings, never crash. Pinned in Tasks 13 and 15.
3. A domain whose TLD has no RDAP server, or no network at all: `names check` must report `unknown` with a warning and exit 0. Pinned in Task 19.
4. A concept whose scorecard fails the gate or whose audit failed: `sheet concepts` must refuse it with exit 2 and name the reason, so nothing unvetted reaches the owner. Pinned in Task 13.
5. PNG exports with a transparent `reverse` or `white` variant: the file must have real alpha (not a white box), and the README must say to preview them on dark. Pinned in Task 15.

class TerritoryFont(BaseModel):
    family: str
    weight: int = 400


class TerritoryColor(BaseModel):

    slug: str
    name: str
    embraces: list[str]
    rejects: list[str]
    fonts: list[TerritoryFont] = Field(min_length=1, max_length=3)
    palette: list[TerritoryColor] = Field(min_length=2, max_length=6)
    shapes: TerritoryShapes = Field(default_factory=TerritoryShapes)


class TerritoriesSpec(BaseModel):

```python
BOARD_W = 1600


def open_file(path: Path, enabled: bool = True) -> bool:
    """Open a file with the macOS `open` command unless disabled by flag or LOGOMAKER_NO_OPEN."""
    if not enabled or os.environ.get("LOGOMAKER_NO_OPEN"):
        return False
    subprocess.Popen(["open", str(path)])
    return True


def _ensure_font(paths: ProjectPaths, family: str) -> None:
    if not fontlib.cached_files(paths, family):
        fontlib.fetch(paths, family)


def shape_specimens(shapes: TerritoryShapes, initial: str, family: str, weight: int) -> list[dict]:
    width = 300 * shapes.stroke
    join = "miter" if shapes.angularity >= 0.5 else "round"
    cap = "butt" if shapes.angularity >= 0.5 else "round"
    tile_radius = 300 * shapes.corner
    bars = [rect_centered(0, y, 300, width) for y in (-100, 0, 100)]
    if shapes.symmetry < 0.5:
        bars = [rect_centered(-30, -100, 240, width), rect_centered(0, 0, 300, width), rect_centered(45, 100, 210, width)]
    specimens = [
        ("ring", ring(0, 0, 150, width)),
        ("chevron", polyline([(-120, -70), (0, 70), (120, -70)], width, cap=cap, join=join)),
        ("tile", rrect_centered(0, 0, 260, 260, tile_radius)),
        ("loop", arc(0, 0, 110, 0, 270 if shapes.symmetry >= 0.5 else 300, width, cap=cap)),
        ("bars", union(*bars)),
        ("initial", glyph(initial, family, weight=weight, cap_height=240)),
    ]
    return [{"label": label, "svg": Mark(symbol=shape).to_svg("black", padding=20)} for label, shape in specimens]


def territories_board(paths: ProjectPaths, slug: str, renderer: Renderer | None = None, open_after: bool = True) -> dict:
    state = registry.load_state(paths, slug)
    folder = paths.brand(slug) / "territories"
    spec_path = folder / "territories.json"
    if not spec_path.is_file():
        raise CliError(f"{spec_path.relative_to(paths.root)} is missing; write the territories spec first")
    spec = TerritoriesSpec.model_validate_json(spec_path.read_text(encoding="utf-8"))
    cards = []
    for t in spec.territories:
        fonts = []
        for f in t.fonts:
            _ensure_font(paths, f.family)
            word = text(state.name, f.family, weight=f.weight, cap_height=200)
            fonts.append({"label": f"{f.family} {f.weight}", "svg": Mark(wordmark=word).to_svg("black", padding=10)})
        palette = []
        for c in t.palette:
            i = color_info(c.hex)
            palette.append({"name": c.name, "hex": i["hex"], "oklch": f"oklch({i['oklch'][0]:.2f} {i['oklch'][1]:.3f} {i['oklch'][2]:.0f})",
                            "text": "#ffffff" if luminance(c.hex) < 0.35 else "#111111"})
        cards.append({"name": t.name, "embraces": t.embraces, "rejects": t.rejects, "fonts": fonts, "palette": palette,
                      "shapes": shape_specimens(t.shapes, state.name[:1].upper() or "A", t.fonts[0].family, t.fonts[0].weight)})
    html = jinja(paths).get_template("sheets/territories.html").render(brand=state.name, territories=cards, width=BOARD_W)
    html_path = folder / "board.html"
    html_path.write_text(html, encoding="utf-8")
    if renderer is None:
        with Renderer() as own:
            png = own.html_to_png(html, BOARD_W, full_page=True)
    else:
        png = renderer.html_to_png(html, BOARD_W, full_page=True)
    (folder / "board.png").write_bytes(png)
    open_file(html_path, open_after)
    return {"board_html": str(html_path), "board_png": str(folder / "board.png"), "territories": [t.slug for t in spec.territories]}
```

```python
CRITERIA = ("appropriate", "distinctive", "simple", "legible", "adaptable", "timeless", "universal", "clear", "congruent", "system")
GATE = ("appropriate", "simple", "legible")


class Scorecard(BaseModel):
    scores: dict[str, int]
    evidence: dict[str, str] = Field(default_factory=dict)
    kill: bool = False
    reason: str = ""

    @model_validator(mode="after")
    def complete_and_in_range(self) -> "Scorecard":
        missing = [c for c in CRITERIA if c not in self.scores]
        if missing:
            raise ValueError(f"scores missing: {', '.join(missing)}")
        bad = {k: v for k, v in self.scores.items() if not 1 <= v <= 5}
        if bad:
            raise ValueError(f"scores must be 1..5: {bad}")
        return self

    @property
    def gate_ok(self) -> bool:
        return not self.kill and all(self.scores[c] >= 4 for c in GATE)

    @property
    def total(self) -> int:
        return sum(self.scores[c] for c in CRITERIA)

CONCEPTS_W = 1800
ALL_ROLES = ("primary", "secondary", "accent", "ink", "paper")


def mockup_context(state, outputs: dict, light: dict | None, dark: dict | None) -> dict:
    warnings: list[str] = []
    if "full-symbol" not in outputs:
        warnings.append("no symbol in this concept: icon, favicon and avatar mockups use the wordmark")
    horizontal = outputs.get("full-horizontal") or outputs["full-lockup"]
    paper_light = light["paper"] if light else "#ffffff"
    paper_dark = dark["paper"] if dark else "#111111"
    tile_fill = light["primary"] if light else "#111111"
    on_tile = {role: paper_light for role in ALL_ROLES}

    def colour(mark, size_w, size_h):
        svg = mark.to_svg("color", palette=light) if light else mark.to_svg("black")
        return fit_box(svg, size_w, size_h)

    def reverse(mark, size_w, size_h):
        svg = mark.to_svg("reverse", palette=dark) if dark else mark.to_svg("white")
        return fit_box(svg, size_w, size_h)

    return {
        "brand": state.name,
        "handle": slugify(state.name).replace("-", ""),
        "paper_light": paper_light, "paper_dark": paper_dark, "primary": tile_fill,
        "primary_dark": dark["primary"] if dark else "#eeeeee", "tile_fill": tile_fill,
        "icon_svg": fit_box(outputs[pick_output(outputs, 60)].to_svg("color", palette=on_tile), 36, 36),
        "favicon_svg": colour(outputs[pick_output(outputs, 16)], 16, 16),
        "header_svg": colour(horizontal, 260, 32),
        "header_dark_svg": reverse(horizontal, 260, 32),
        "avatar_svg": colour(outputs[pick_output(outputs, 80)], 48, 48),
        "card_svg": colour(horizontal, 200, 60),
        "sign_svg": fit_box(outputs["full-lockup"].to_svg("white"), 360, 110),
        "neighbours": mockuplib.neighbour_icons(60),
        "tabs": ["Inbox", "Calendar"],
        "warnings": warnings,
    }


def _vetted(folder: Path, name: str) -> Scorecard:
    audit_file = folder / "audit.json"
    if not audit_file.is_file():
        raise CliError(f"concept {name}: audit.json is missing; run logomaker concept run first")
    if not AuditResult.model_validate_json(audit_file.read_text(encoding="utf-8")).ok:
        raise CliError(f"concept {name}: audit has failures; fix or kill it before presenting")
    card_file = folder / "scorecard.json"
    if not card_file.is_file():
        raise CliError(f"concept {name}: scorecard.json is missing; score it before presenting")
    card = Scorecard.model_validate_json(card_file.read_text(encoding="utf-8"))
    if not card.gate_ok:
        raise CliError(f"concept {name}: scorecard fails the gate (appropriate, simple and legible must be 4 or more, and not killed)")
    return card


def concepts_sheet(paths: ProjectPaths, slug: str, concept_names: list[str], renderer: Renderer | None = None, open_after: bool = True) -> dict:
    state = registry.load_state(paths, slug)
    env = jinja(paths)
    cards = []
    for name in concept_names:
        folder = conceptlib.concept_folder(paths, slug, name)
        card = _vetted(folder, name)
        module = conceptlib.load_concept(folder)
        outputs = conceptlib.build_outputs(conceptlib.build_tiers(module))
        light, dark = _palettes(folder)
        ctx = mockup_context(state, outputs, light, dark)
        lockup = outputs["full-lockup"]
        cards.append({
            "name": name, "idea": module.meta.idea, "total": card.total, "scores": card.scores,
            "black": fit_box(lockup.to_svg("black"), 320, 200), "white": fit_box(lockup.to_svg("white"), 320, 200),
            "ladder": [{"size": s, "svg": fit_box(outputs[pick_output(outputs, s)].to_svg("black"), s, s)} for s in (128, 64, 32, 16)],
            "warnings": ctx["warnings"],
            "mockups": [mockuplib.render_mockup(env, kind, ctx) for kind in mockuplib.MOCKUPS],
        })
    html = env.get_template("sheets/concepts.html").render(brand=state.name, concepts=cards, width=CONCEPTS_W)
    sheets_dir = paths.brand(slug) / "sheets"
    sheets_dir.mkdir(exist_ok=True)
    html_path = sheets_dir / "concepts.html"
    html_path.write_text(html, encoding="utf-8")
    if renderer is None:
        with Renderer() as own:
            png = own.html_to_png(html, CONCEPTS_W, full_page=True)
    else:
        png = renderer.html_to_png(html, CONCEPTS_W, full_page=True)
    (sheets_dir / "concepts.png").write_bytes(png)
    open_file(html_path, open_after)
    return {"sheet_html": str(html_path), "sheet_png": str(sheets_dir / "concepts.png"), "concepts": concept_names}
```

"""Export package builder (spec section 9). Each group writes files and registers manifest entries."""
from __future__ import annotations

import hashlib
import io
import json
import re
import shutil
from pathlib import Path

from PIL import Image

from . import registry
from .color import enrich_tokens, load_tokens
from .kit import concept as conceptlib
from .kit.mark import Mark
from .output import CliError
from .paths import ProjectPaths
from .render import Renderer, svg_size, with_size
from .schemas import Manifest, ManifestEntry, now_iso
from .sheets import jinja

VARIANT_NAMES = ("primary", "horizontal", "stacked", "wordmark", "mark", "small", "favicon")
COLOURS = ("color", "reverse", "black", "white", "mono")
PNG_VARIANTS = ("primary", "horizontal", "stacked", "wordmark", "mark")
PNG_COLOURS = ("color", "reverse", "black", "white")
PNG_SIZES = (256, 512, 1024, 2048)
MASK_SAFE = 289
REQUIRED_FINAL = ("concept.py", "params.json", "tokens.json", "rationale.md")
_VIEWBOX = re.compile(r'viewBox="([^"]+)"')


def square_viewbox(svg: str, fraction: float) -> str:
    """Rewrite the viewBox to a square in which the artwork's longer side spans `fraction` of the side."""
    match = _VIEWBOX.search(svg)
    x, y, w, h = (float(v) for v in match.group(1).split())
    side = max(w, h) / fraction
    nx, ny = x + w / 2 - side / 2, y + h / 2 - side / 2
    return svg.replace(match.group(0), f'viewBox="{nx:g} {ny:g} {side:g} {side:g}"', 1)


class PackageBuilder:
    GROUPS: list[str] = ["masters", "pngs", "flattened", "favicons"]

    def __init__(self, paths: ProjectPaths, slug: str, paper: str = "letter"):
        self.paths, self.slug, self.paper = paths, slug, paper
        self.state = registry.load_state(paths, slug)
        self.brand_dir = paths.brand(slug)
        self.final = self.brand_dir / "final"
        self.pkg = self.brand_dir / "package"
        for name in REQUIRED_FINAL:
            if not (self.final / name).is_file():
                raise CliError(f"final/{name} is missing; finish refinement (and write final/rationale.md) before packaging")
        self.module = conceptlib.load_concept(self.final)
        self.outputs = conceptlib.build_outputs(conceptlib.build_tiers(self.module))
        self.tokens = enrich_tokens(load_tokens(self.final / "tokens.json"))
        self.light = self.tokens.palette("light")
        self.dark = self.tokens.palette("dark")
        self.name = registry.slugify(self.state.name)
        self.entries: list[ManifestEntry] = []
        self.warnings: list[str] = []
        self.renderer: Renderer | None = None

    # ---- helpers -------------------------------------------------------
    def warn(self, message: str) -> None:
        if message not in self.warnings:
            self.warnings.append(message)

    def mark(self, variant: str) -> Mark | None:
        primary = "full-horizontal" if self.state.primary_lockup == "horizontal" else "full-stacked"
        table = {
            "primary": [primary, "full-lockup"], "horizontal": ["full-horizontal", "full-lockup"], "stacked": ["full-stacked", "full-lockup"],
            "wordmark": ["full-wordmark"], "mark": ["full-symbol", "full-lockup"], "small": ["small", "full-symbol", "full-lockup"],
            "favicon": ["favicon", "small", "full-symbol", "full-lockup"],
        }
        for name in table[variant]:
            if name in self.outputs:
                if name != table[variant][0]:
                    self.warn(f"{variant} falls back to {name} (draw a dedicated tier for better small sizes)")
                return self.outputs[name]
        self.warn(f"{variant} skipped: the concept has no part for it")
        return None

    def svg(self, variant: str, colour: str) -> str | None:
        mark = self.mark(variant)
        if mark is None:
            return None
        return mark.to_svg(colour, palette=self.dark if colour == "reverse" else self.light)

    def write(self, relpath: str, data: bytes | str, purpose: str, width: int | None = None, height: int | None = None) -> Path:
        target = self.pkg / relpath
        target.parent.mkdir(parents=True, exist_ok=True)
        raw = data.encode("utf-8") if isinstance(data, str) else data
        target.write_bytes(raw)
        self.entries.append(ManifestEntry(path=relpath, purpose=purpose, width=width, height=height, sha256=hashlib.sha256(raw).hexdigest(), bytes=len(raw)))
        return target

    def png_bytes(self, svg: str, long_edge: int) -> bytes:
        w, h = svg_size(svg, long_edge)
        if h > w:
            w, h = max(1, round(long_edge * w / h)), long_edge
        return self.renderer.svg_to_png(svg, w, h)

    def render_fit(self, svg: str, box_w: int, box_h: int, background: str | None = None) -> bytes:
        _, h_at_w = svg_size(svg, box_w)
        w, h = (box_w, h_at_w) if h_at_w <= box_h else (max(1, round(box_w * box_h / h_at_w)), box_h)
        return self.renderer.svg_to_png(svg, w, h, background=background)

    @staticmethod
    def compose(width: int, height: int, background: str | None, art: bytes | None = None, box: tuple[int, int, int, int] | None = None) -> bytes:
        canvas = Image.new("RGBA", (width, height), background if background else (0, 0, 0, 0))
        if art is not None:
            image = Image.open(io.BytesIO(art)).convert("RGBA")
            bx, by, bw, bh = box or (0, 0, width, height)
            scale = min(bw / image.width, bh / image.height, 1.0)
            if scale < 1.0:
                image = image.resize((max(1, round(image.width * scale)), max(1, round(image.height * scale))), Image.LANCZOS)
            canvas.alpha_composite(image, (bx + (bw - image.width) // 2, by + (bh - image.height) // 2))
        out = io.BytesIO()
        canvas.save(out, format="PNG")
        return out.getvalue()

    def _png_image(self, svg: str, long_edge: int) -> Image.Image:
        return Image.open(io.BytesIO(self.png_bytes(svg, long_edge))).convert("RGBA")

    # ---- groups --------------------------------------------------------
    def masters(self) -> None:
        for variant in VARIANT_NAMES:
            for colour in COLOURS:
                svg = self.svg(variant, colour)
                if svg is not None:
                    self.write(f"{self.name}-{variant}-{colour}.svg", svg, f"{variant} lockup, {colour} variant, vector master")

    def pngs(self) -> None:
        for variant in PNG_VARIANTS:
            for colour in PNG_COLOURS:
                svg = self.svg(variant, colour)
                if svg is None:
                    continue
                for size in PNG_SIZES:
                    png = self.png_bytes(svg, size)
                    image = Image.open(io.BytesIO(png))
                    self.write(f"{self.name}-{variant}-{colour}-{size}.png", png, f"{variant} {colour} transparent PNG, long edge {size}", image.width, image.height)

    def flattened(self) -> None:
        svg = self.svg("primary", "color")
        if svg is None:
            return
        image = self._png_image(svg, 2048)
        flat = Image.new("RGB", image.size, self.light["paper"])
        flat.paste(image, mask=image.split()[3])
        out = io.BytesIO()
        flat.save(out, format="JPEG", quality=92)
        self.write(f"{self.name}-primary-color-white-bg-2048.jpg", out.getvalue(), "primary on paper background, flattened JPEG", *image.size)
        out = io.BytesIO()
        image.save(out, format="WEBP", quality=90)
        self.write(f"{self.name}-primary-color-2048.webp", out.getvalue(), "primary colour transparent WebP", *image.size)

    def favicons(self) -> None:
        fav = self.mark("favicon")
        small = self.mark("small")
        if fav is None or small is None:
            return
        fav_svg = square_viewbox(fav.to_svg("color", palette=self.light), 0.9)

```python
"""Vector PDF and EPS from kit paths via reportlab. CMYK is a naive conversion; the README tells printers to confirm."""
from __future__ import annotations

from pathlib import Path

from reportlab.graphics import renderPS
from reportlab.graphics.shapes import Drawing
from reportlab.graphics.shapes import Path as RLPath
from reportlab.lib.colors import CMYKColor, Color
from reportlab.pdfgen import canvas as rl_canvas

from .color import info
from .kit.mark import Mark, fill_for

UNIT_PT = 0.25  # 1000 kit units = 250 pt (about 88 mm)
FILL_NON_ZERO = 1


def _colour(hex_value: str, mode: str):
    if hex_value == "currentColor":
        raise ValueError("the mono variant cannot be printed; use color, reverse, black or white")
    i = info(hex_value)
    if mode == "cmyk":
        c, m, y, k = (v / 100 for v in i["cmyk"])
        return CMYKColor(c, m, y, k)
    r, g, b = (v / 255 for v in i["rgb"])
    return Color(r, g, b)


def _fills(mark: Mark, variant: str, palette: dict | None, mode: str):
    return [(p.shape, _colour(fill_for(p.role, variant, palette), mode)) for p in mark.layout()]


def mark_to_pdf(mark: Mark, out: Path, variant: str = "color", palette: dict | None = None, mode: str = "rgb") -> Path:
    fills = _fills(mark, variant, palette, mode)
    b = mark.bounds()
    c = rl_canvas.Canvas(str(out), pagesize=(b.w * UNIT_PT, b.h * UNIT_PT))
    c.scale(UNIT_PT, UNIT_PT)
    c.translate(-b.x0, -b.y0)
    for shape, colour in fills:
        c.setFillColor(colour)
        path = c.beginPath()
        for kind, pts in shape.commands():
            if kind == "M":
                path.moveTo(*pts[0])
            elif kind == "L":
                path.lineTo(*pts[0])
            elif kind == "C":
                path.curveTo(*pts[0], *pts[1], *pts[2])
            elif kind == "Z":
                path.close()
        c.drawPath(path, stroke=0, fill=1, fillMode=FILL_NON_ZERO)
    c.showPage()
    c.save()
    return out


def mark_to_eps(mark: Mark, out: Path, variant: str = "color", palette: dict | None = None, mode: str = "cmyk") -> Path:
    fills = _fills(mark, variant, palette, mode)
    b = mark.bounds()
    drawing = Drawing(b.w * UNIT_PT, b.h * UNIT_PT)
    for shape, colour in fills:
        rp = RLPath(fillColor=colour, strokeColor=None, strokeWidth=0)
        rp.fillMode = FILL_NON_ZERO
        for kind, pts in shape.commands():
            scaled = [((x - b.x0) * UNIT_PT, (y - b.y0) * UNIT_PT) for x, y in pts]
            if kind == "M":
                rp.moveTo(*scaled[0])
            elif kind == "L":
                rp.lineTo(*scaled[0])
            elif kind == "C":
                rp.curveTo(*scaled[0], *scaled[1], *scaled[2])
            elif kind == "Z":
                rp.closePath()
        drawing.add(rp)
    renderPS.drawToFile(drawing, str(out))
    return out
```

def verify(paths: ProjectPaths, slug: str) -> dict:
    pkg = paths.brand(slug) / "package"
    manifest_path = pkg / "manifest.json"
    if not manifest_path.is_file():
        raise CliError("package/manifest.json is missing; run logomaker package build first", code=EXIT_VERIFY)
    manifest = Manifest.model_validate_json(manifest_path.read_text(encoding="utf-8"))
    readme = (pkg / "README.md").read_text(encoding="utf-8") if (pkg / "README.md").is_file() else ""
    problems: list[str] = []
    for entry in manifest.files:
        path = pkg / entry.path
        if not path.is_file():
            problems.append(f"{entry.path} is missing")
            continue
        raw = path.read_bytes()
        if not raw:
            problems.append(f"{entry.path} is empty")
            continue
        if hashlib.sha256(raw).hexdigest() != entry.sha256:
            problems.append(f"{entry.path} changed since the build")
        if entry.width and path.suffix.lower() in (".png", ".jpg", ".jpeg", ".webp"):
            size = Image.open(io.BytesIO(raw)).size
            if size != (entry.width, entry.height):
                problems.append(f"{entry.path} is {size[0]}x{size[1]}, expected {entry.width}x{entry.height}")
        if f"`{entry.path}`" not in readme:
            problems.append(f"README.md does not list {entry.path}")
    result = {"ok": not problems, "problems": problems, "files": len(manifest.files), "checked_at": now_iso()}
    (pkg / "verify.json").write_text(json.dumps(result, indent=2) + "\n", encoding="utf-8")
    return result

def fake_transport(registered: set[str], github_taken: set[str], fail: set[str] = frozenset()):

            if domain in fail:

```python
"""Name analysis (sound and letterform features) and availability checks over RDAP, whois and the GitHub API."""
from __future__ import annotations

import re
import subprocess
import unicodedata

import httpx

FRONT, BACK, NEUTRAL = set("ie"), set("ou"), set("a")
VOWELS = FRONT | BACK | NEUTRAL | {"y"}
ROUND, DIAGONAL, CLOSED = set("ocgsq"), set("avwkxz"), set("abdopqr")
RDAP = "https://rdap.org/domain/{domain}"
GITHUB_USER = "https://api.github.com/users/{label}"
GITHUB_ORG = "https://api.github.com/orgs/{label}"
NOT_FOUND_MARKERS = ("no match", "not found", "no data found", "no entries found", "nothing found", "is free", "available")


def domain_label(name: str) -> str:
    text = unicodedata.normalize("NFKD", name).encode("ascii", "ignore").decode().lower()
    return re.sub(r"[^a-z0-9]", "", text)


def analyze(name: str) -> dict:
    letters = re.sub(r"[^a-z]", "", name.lower())
    groups = re.findall(r"[aeiouy]+", letters)
    if letters.endswith("e") and len(groups) > 1 and letters[-2] not in VOWELS:
        groups = groups[:-1]  # silent e: "keystone" has two syllables, not three
    front = sum(ch in FRONT for ch in letters)
    back = sum(ch in BACK for ch in letters)
    neutral = sum(ch in NEUTRAL for ch in letters)
    runs = [len(r) for r in re.findall(r"[^aeiouy]+", letters)]
    initial = letters[:1]
    doubles = [m.group(0) for m in re.finditer(r"([a-z])\1", letters)]
    return {
        "name": name,
        "letters": letters,
        "length": len(letters),
        "syllables": max(1, len(groups)) if letters else 0,
        "vowels": {"front": front, "back": back, "neutral": neutral, "front_ratio": round(front / max(1, front + back), 2)},
        "initial": {"letter": initial, "round": initial in ROUND, "diagonal": initial in DIAGONAL, "closed_counter": initial in CLOSED, "l_warning": initial == "l"},
        "double_letters": doubles,
        "camel_case": bool(re.search(r"[a-z][A-Z]", name)),
        "has_digits": bool(re.search(r"\d", name)),
        "max_consonant_run": max(runs, default=0),
        "pronounceable": max(runs, default=0) <= 3 and bool(groups),
    }


def default_transport() -> httpx.BaseTransport | None:
    return None


def _whois(domain: str) -> tuple[str, str]:
    try:
        out = subprocess.run(["whois", domain], capture_output=True, text=True, timeout=10)
    except (OSError, subprocess.SubprocessError):
        return "unknown", "none"
    text = (out.stdout + out.stderr).lower()
    if any(marker in text for marker in NOT_FOUND_MARKERS):
        return "available", "whois"
    if "domain name:" in text or "registrar:" in text:
        return "registered", "whois"
    return "unknown", "whois"


def check_domain(domain: str, client: httpx.Client) -> dict:
    try:
        resp = client.get(RDAP.format(domain=domain))
        if resp.status_code == 200:
            return {"domain": domain, "status": "registered", "method": "rdap"}
        if resp.status_code == 404:
            return {"domain": domain, "status": "available", "method": "rdap"}
    except httpx.HTTPError:
        pass
    status, method = _whois(domain)
    return {"domain": domain, "status": status, "method": method}


def check_github(label: str, client: httpx.Client) -> dict:
    out = {}
    for key, url in (("user", GITHUB_USER), ("org", GITHUB_ORG)):
        try:
            code = client.get(url.format(label=label)).status_code
            out[key] = "available" if code == 404 else "taken" if code == 200 else "unknown"
        except httpx.HTTPError:
            out[key] = "unknown"
    return out


def check(name: str, tlds: tuple[str, ...] = ("com", "co", "io", "ai", "app", "dev"), handles: tuple[str, ...] = ("github",),
          transport: httpx.BaseTransport | None = None) -> dict:
    label = domain_label(name)
    result = {"name": name, "label": label, "domains": [], "handles": {}, "warnings": []}
    if not label:
        result["warnings"].append("name has no letters or digits to check")
        return result
    with httpx.Client(timeout=10, follow_redirects=True, headers={"User-Agent": "logomaker", "Accept": "application/json"}, transport=transport or default_transport()) as client:
        for tld in tlds:
            row = check_domain(f"{label}.{tld}", client)
            if row["status"] == "unknown":
                result["warnings"].append(f"{row['domain']}: could not determine availability (no RDAP answer and whois gave nothing)")
            result["domains"].append(row)
        if "github" in handles:
            result["handles"]["github"] = check_github(label, client)
            if "unknown" in result["handles"]["github"].values():
                result["warnings"].append("github: rate limited or unreachable; try again later")
    return result
```

@main.group("names")
def names_group() -> None:
    """Name analysis and availability checks."""


@names_group.command("analyze")
@click.argument("name")
@command
def names_analyze(name: str) -> None:
    emit(nameslib.analyze(name))


@names_group.command("check")
@click.argument("name")
@click.option("--tlds", default="com,co,io,ai,app,dev")
@click.option("--handles", default="github")
@command
def names_check(name: str, tlds: str, handles: str) -> None:
    result = nameslib.check(name, tuple(t.strip() for t in tlds.split(",") if t.strip()), tuple(h.strip() for h in handles.split(",") if h.strip()))
    emit(result, warnings=result["warnings"])
